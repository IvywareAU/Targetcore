# TargetCore -> libtargetcore.so (Linux) / TargetCore.dll (Windows/MSVC via CMake).
# Enabled when MSCS_BUILD_LIBS=ON. On Windows the CMake build mirrors the (2022).vcxproj
# (CMake-on-Windows parity): dynamic MFC, native IOCP (no io_uring
# shim), the CNG/BCrypt crypto backend. The vcxproj stays authoritative until this path
# is signed off.
#
# Sources COMMON to both platforms: the pumps (P2Pwin32), object model, all four transports
# (Dmx / Wsa / Pipe / 232), P2PeerService (SCM shimmed to no-op on Linux), and the flat C
# API (base wchar_t TargetCore_c.cpp + the UTF-8 _u8 twins for the Panama/jextract bridge,
# §4.2/§6.2).
#
# Buint (the hand-rolled big-integer type) was removed on 2026-08-24. It was the last piece
# of the legacy DH/Rijndael apparatus deleted with DHKeyXChanger.cpp and Rijndael.cpp: by
# then its only callers anywhere in the tree were its own Buint_Test_* self-tests, and its
# Random() still seeded with srand(time(NULL)) -- so a grep for srand over a repository
# whose live crypto is CNG/OpenSSL kept finding it and kept having to be explained away.
# SECURITY_REVIEW.md records it as dead rather than exploitable. The class was exported, so
# this drops its symbols from the DLL export table (see the versioning policy on the export check).
#
# The flat C API was briefly removed on the grounds that no in-tree consumer imported one
# of its 83 symbols, with the sources parked in MSCS_JavaBindings\TargetCore\native\. It is
# back here, and THIS is the authoritative copy -- the Java tree's bindings are generated by
# running jextract over this header, so it has to be the one that is compiled. The absence
# of an in-tree consumer is a fact about this repository, not about the API: the consumer is
# out-of-tree by construction, which is what an FFI surface is for.
#
# There is no second copy any more. MSCS_JavaBindings\TargetCore\native\ held a byte-identical
# one; it was deleted on 2026-08-14 (MSCS_JavaBindings d5bd3fa) after checking it matched, and
# that tree's jextract step now reads THIS header. Worth knowing why it mattered: the copy
# COMMITTED over there predated the handle registry, so the bindings tree documented a library
# several fixes older than the one it loads, and nothing would have reported the disagreement.
#
# The practical consequence of building it again: every symbol is exported from the DLL/.so,
# so these entry points are a supported surface and the M1 handle registry in
# TargetCore_c.cpp is load-bearing rather than dormant.
#
# The legacy hand-rolled crypto (DHKeyXChanger, Rijndael) has been REMOVED - both were
# dormant, neither was ever constructed, and DHKeyXChanger carried the unbounded pre-auth
# bignum parse filed as SECURITY_REVIEW M7. The live crypto is P2PCngCrypto (CNG) /
# P2PCngCrypto_openssl (OpenSSL) driving P2PeerioGcm.
#
# P2PIdentityStore.cpp is COMMON although it is a crypto file: the container format, the
# atomic write and the text parsing are platform-independent, and only protect/unprotect
# and the file-permission call differ (#ifdef inside the TU). It includes no stdafx.h.
#
# P2PAuthLogin.cpp is COMMON for the same reason and more strongly: the login
# transcript and block layout ARE the wire, so a byte of drift between the CNG
# and OpenSSL backends is a login that works only within one operating system.
# One TU, no #ifdef except the self-test's temp directory, no stdafx.h.
set(TARGETCORE_SOURCES_COMMON
    P2PAuthLogin.cpp
    P2PIdentityStore.cpp
    P2PeerService.cpp
    P2Peer.cpp
    P2PeerCon.cpp
    P2PeerCon232.cpp
    P2PeerConDmx.cpp
    P2PeerConPipe.cpp
    P2PeerConWsa.cpp
    P2PeerEvents.cpp
    P2PeerExplorer.cpp
    P2PeerHub.cpp
    P2Peerio.cpp
    P2PeerioBSTR.cpp
    P2PeerioGcm.cpp
    P2PeerioDmx.cpp
    P2PeerSeal.cpp
    P2PeerMsg.cpp
    P2PeerMsgQue.cpp
    P2PeerTarget.cpp
    TargetCoreLog.cpp
    P2Pwin32.cpp
    WSA_Ext.cpp
    TargetCore_c.cpp
    TargetCore_c_u8.cpp)

if(WIN32)
    # Windows/MSVC: the CNG/BCrypt crypto backend (P2PCngCrypto.cpp, body #ifdef _WIN32) and
    # the MFC-extension-DLL entry (dllmain). Native IOCP lives in P2Pwin32/P2PeerCon, so no
    # io_uring shim. MFC dynamic (UseOfMfc=Dynamic); CMAKE_MFC_FLAG must precede add_library().
    set(CMAKE_MFC_FLAG 2)
    # TargetCore.rc carries VS_VERSION_INFO (company, product, version, copyright) and the
    # IDS_APP_TITLE string. Without it a CMake-built DLL has no .rsrc section at all, so it
    # reports no version to Explorer or to an installer while the vcxproj-built one does.
    # project() at the top level declares LANGUAGES CXX only, so RC must be enabled here
    # before a .rc can be attached to a target.
    enable_language(RC)
    set(TARGETCORE_SOURCES ${TARGETCORE_SOURCES_COMMON}
        P2PCngCrypto.cpp
        dllmain.cpp
        TargetCore.rc)
    # Excluded (as in the vcxproj-vs-CMake split): stdafx.cpp is the classic /Yc PCH TU; this
    # build compiles each TU's own #include "stdafx.h" without a target PCH (see note below).
else()
    # Linux: the OpenSSL crypto backend (p2pcng AES-256-GCM / ECDH P-256 / HKDF / HMAC / RNG,
    # wire-compatible with CNG, §6.2 Risk #5) and the io_uring implementation of the
    # p2p_iocp_* hooks. dllmain.cpp is excluded (§4.3).
    set(TARGETCORE_SOURCES ${TARGETCORE_SOURCES_COMMON}
        P2PCngCrypto_openssl.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/../Platform/p2piocp.cpp)
endif()

add_library(targetcore SHARED ${TARGETCORE_SOURCES})
target_compile_features(targetcore PUBLIC cxx_std_23)
target_include_directories(targetcore PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/../Msgcore)
# msgcore is a real, exported dependency: the installed .so/.dll needs the
# installed libmsgcore beside it to load, and an installed consumer that wants a
# handle type from Msgcore_c.h needs its header. p2pplatform is NOT -- it is a
# build-time include-path shim that publishes nothing, so it is wrapped in
# $<BUILD_INTERFACE:> (the full reasoning is in Msgcore/CMakeLists.txt beside
# the identical wrap). The build tree is unchanged either way; only the export
# is different, and only the export was ever wrong.
target_link_libraries(targetcore PUBLIC msgcore $<BUILD_INTERFACE:p2pplatform>)
# Where an INSTALLED consumer finds TargetCore_c.h. Empty in the build tree.
target_include_directories(targetcore INTERFACE $<INSTALL_INTERFACE:include>)
target_compile_definitions(targetcore PRIVATE TargetCore_EXPORTS _UNICODE UNICODE)

if(WIN32)
    # Match the vcxproj preprocessor set (CMAKE_MFC_FLAG adds _AFXDLL + the shared runtime;
    # _AFXEXT marks the MFC extension DLL).
    target_compile_definitions(targetcore PRIVATE WIN32 _WINDOWS _AFXEXT)
    # See the identical note in Msgcore/CMakeLists.txt: the target is lowercase because CMake
    # targets are identifiers, the artifact is TargetCore.dll because that is what the vcxproj
    # ships and what every consumer resolves.
    set_target_properties(targetcore PROPERTIES OUTPUT_NAME TargetCore)
    # Nothing stages this DLL into bin\<Config><Arch>, deliberately. See the block headed
    # "WHY THERE IS NO STAGING STEP HERE" in the top-level CMakeLists.txt before adding one.
    # The vcxproj system deps (Winsock, COM support for _bstr_t/_variant_t, Propsys) plus the
    # CNG crypto libs the BCrypt backend needs: bcrypt (AES-GCM / HKDF / HMAC / RNG) and
    # ncrypt (ECDH P-256).
    # crypt32/advapi32 are for P2PIdentityStore.cpp (DPAPI + the SDDL DACL on the
    # identity file). The vcxproj gets them from #pragma comment(lib) in that TU;
    # naming them here keeps the CMake link line self-describing.
    target_link_libraries(targetcore PRIVATE
        ws2_32 MsWsock Propsys bcrypt ncrypt crypt32 advapi32
        comsuppw$<$<CONFIG:Debug>:d>)
else()
    # Legacy TUs need -fpermissive (the MSVC-lax member-pointer / address-of forms); harmless
    # on the clean C++23 p2piocp.cpp.
    target_compile_options(targetcore PRIVATE -fpermissive)

    # liburing for p2piocp.cpp (io_uring). Required on Linux.
    #
    # $<BUILD_INTERFACE:> on both this and OpenSSL below: they are implementation
    # dependencies of a SHARED library. libtargetcore.so records them in its own
    # DT_NEEDED, so a consumer that links the installed library needs neither
    # liburing-dev nor libssl-dev to build against it -- and an exported
    # OpenSSL::Crypto would have forced find_dependency(OpenSSL) on every
    # consumer, which is a development package demanded at build time for a
    # dependency that is already satisfied at run time. In-tree propagation is
    # unchanged: BUILD_INTERFACE evaluates to exactly what was there before.
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBURING REQUIRED liburing)
    target_include_directories(targetcore PRIVATE ${LIBURING_INCLUDE_DIRS})
    target_link_libraries(targetcore PUBLIC $<BUILD_INTERFACE:${LIBURING_LIBRARIES}>)

    # OpenSSL 3 (libcrypto) for P2PCngCrypto_openssl.cpp (§6.2). Required on Linux.
    find_package(OpenSSL REQUIRED)
    target_link_libraries(targetcore PUBLIC $<BUILD_INTERFACE:OpenSSL::Crypto>)
endif()

# No target_precompile_headers: the legacy TUs #include "stdafx.h" themselves, and a
# target-wide PCH would wrongly prepend stdafx.h to the clean p2piocp.cpp (Linux).

# ---------------------------------------------------------------------------
# The DebugLib/ReleaseLib shape (MSCS_BUILD_STATIC_LIBS, off by default).
#
# Taken from TargetCore(2022).vcxproj rather than assumed. Those configurations are
# ConfigurationType=StaticLibrary, UseOfMfc=Dynamic (so CMAKE_MFC_FLAG 2 still applies),
# preprocessor WIN32;_WINDOWS;_DEBUG;Msgcore_STATIC;TargetCore_STATIC -- no TargetCore_EXPORTS
# and no _AFXEXT -- and, the one difference in the TU list, they set ExcludedFromBuild on
# dllmain.cpp in all four Lib configurations. That is the part a mirror gets wrong by
# defaulting: an archive has no DllMain, and leaving it in links a second one into whatever
# DLL eventually swallows this library. TargetCore.rc is NOT excluded there, so it stays here
# too, even though a version resource in an archive does nothing on its own.
#
# msgcore_static, not msgcore: mixing an archive built with Msgcore_STATIC against the import
# library of Msgcore.dll gives one definition compiled two ways.
# ---------------------------------------------------------------------------
if(MSCS_BUILD_STATIC_LIBS)
    set(TARGETCORE_STATIC_SOURCES ${TARGETCORE_SOURCES})
    list(REMOVE_ITEM TARGETCORE_STATIC_SOURCES dllmain.cpp)
    add_library(targetcore_static STATIC ${TARGETCORE_STATIC_SOURCES})
    target_compile_features(targetcore_static PUBLIC cxx_std_23)
    target_include_directories(targetcore_static PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}/../Msgcore)
    target_link_libraries(targetcore_static PUBLIC msgcore_static p2pplatform)
    target_compile_definitions(targetcore_static PUBLIC TargetCore_STATIC)
    target_compile_definitions(targetcore_static PRIVATE _UNICODE UNICODE)
    if(WIN32)
        target_compile_definitions(targetcore_static PRIVATE WIN32 _WINDOWS)
        target_link_libraries(targetcore_static PUBLIC
            ws2_32 MsWsock Propsys bcrypt ncrypt crypt32 advapi32
            comsuppw$<$<CONFIG:Debug>:d>)
    else()
        target_compile_options(targetcore_static PRIVATE -fpermissive)
        target_include_directories(targetcore_static PRIVATE ${LIBURING_INCLUDE_DIRS})
        target_link_libraries(targetcore_static PUBLIC ${LIBURING_LIBRARIES} OpenSSL::Crypto)
    endif()

    # ---------------------------------------------------------------------
    # p2p_staticlink -- the gate for the archives above.
    #
    # An ARCHIVED target is barely tested by being built. The compiler has seen
    # every TU, but nothing has resolved a symbol across the archive boundary or
    # run a line of it, and both defects found on 2026-08-22 were in exactly
    # that kind of never-linked path. static-check/static_consumer.cpp links
    # both archives and runs the two paths that were broken. Its header comment
    # says what a pass here does and does not mean.
    #
    # Registered here rather than in MscsUnitTests because the archives are
    # defined here and only exist when MSCS_BUILD_STATIC_LIBS is ON, and because
    # this must keep working in a TargetCore + Msgcore checkout with no test
    # repository in it -- the same reasoning as p2p_installtree below.
    add_executable(p2p_staticlink static-check/static_consumer.cpp)
    target_compile_features(p2p_staticlink PRIVATE cxx_std_23)
    target_include_directories(p2p_staticlink PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}/../Msgcore)
    # Msgcore_STATIC / TargetCore_STATIC arrive from the archives' PUBLIC
    # compile definitions rather than being repeated here, and the consumer
    # #errors out if they are missing -- so a build system that wired the SHARED
    # libraries in by mistake fails loudly instead of measuring the wrong thing.
    target_link_libraries(p2p_staticlink PRIVATE targetcore_static msgcore_static)
    if(WIN32)
        target_compile_definitions(p2p_staticlink PRIVATE WIN32 _WINDOWS _UNICODE UNICODE)
    else()
        target_compile_options(p2p_staticlink PRIVATE -fpermissive)
    endif()
    # No loader path: the whole point is that an archive needs no DLL beside it.
    # If this ever needs one, the link was not static.
    add_test(NAME p2p_staticlink COMMAND p2p_staticlink)
    set_tests_properties(p2p_staticlink PROPERTIES TIMEOUT 120)
endif()

# Installation
#
# Until 2026-08-16 this tree had NO install() rule of any kind, in any of the
# three repositories: `cmake --install` produced nothing, so every consumer
# staged binaries by hand out of the build tree and there was no answer to
# "where does the artifact go". That is what these rules fix, and it is worth
# being exact about what they do and do not claim.
#
# On 2026-08-19 (Stage 5 step 14) it was finished. What was
# here staged a library that could not load and a header that could not be
# included; both are below, named, because an install tree that is wrong in a
# way nothing checks is worse than none at all.
#
# WHAT IS INSTALLED: the shared library, TargetCore_c.h, TargetCore_version.h,
# and the licence pair -- and, from the sibling repositories, the same set for
# Msgcore.
#
# TargetCore_version.h WAS MISSING and that made the staged header useless.
# TargetCore_c.h says it is self-contained and it very nearly is, but it opens
# with #include "TargetCore_version.h" (deliberately -- an FFI consumer must be
# able to test what it compiled against). A staged header that does not
# preprocess is not a shipped header, and nothing before the consumer gate below
# would ever have said so.
#
# TargetCore_c.h is still the ONLY header here that a stranger can include. It
# is the surface the Java/Panama bindings are generated from, and its 83 symbols
# are exported from the DLL and the .so alike. The C++ headers are deliberately
# NOT installed: every one of them reaches stdafx.h, which reaches ../Platform
# and MFC, so installing them would publish a surface that cannot compile
# anywhere outside this solution. Shipping a header that cannot be included is
# worse than shipping none -- which is precisely the mistake the version header
# had already made, in miniature.
#
# THE find_package() QUESTION, which this file previously declined. It said a
# config package was impossible because targetcore links msgcore and p2pplatform
# PUBLIC and "exporting targetcore without them fails at generate time". The
# observation is correct and the conclusion did not follow, because the two
# dependencies are not the same kind of thing:
#
#   * msgcore is a REAL dependency. It ships. So it is exported -- from its own
#     repository, by its own CMakeLists.txt, as Msgcore::msgcore -- and this
#     package declares find_dependency(Msgcore). No file reaches into another
#     repository's targets; each exports what it owns, which is the arrangement
#     the repository split already implies.
#   * p2pplatform ships NOTHING. It is an INTERFACE library holding an include
#     directory. Wrapping the link in $<BUILD_INTERFACE:> removes it from the
#     export without touching the build, and Platform/CMakeLists.txt needs no
#     change at all.
#
# So find_package(TargetCore) works, and the sibling comes with it. That the
# answer was three genex wraps and one find_dependency is the reason it should
# not have been deferred; the cost of the wrong answer was every consumer
# hand-staging binaries out of a build tree.
#
# WHAT IS STILL TRUE: the installed library is loadable only with the sibling
# runtime beside it. That has not changed and cannot -- it is what linking
# means. What changed is that `cmake --install` now PUTS it there, and
# p2p_installtree (below) fails if it does not.
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

# Read from the header, not written twice. TargetCore_version.h calls itself the
# single source of version identity and this file must not become a second place
# the number can be wrong.
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/TargetCore_version.h _pm_ver_major
     REGEX "^#define[ \t]+TARGETCORE_VERSION_MAJOR")
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/TargetCore_version.h _pm_ver_minor
     REGEX "^#define[ \t]+TARGETCORE_VERSION_MINOR")
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/TargetCore_version.h _pm_ver_patch
     REGEX "^#define[ \t]+TARGETCORE_VERSION_PATCH")
string(REGEX REPLACE "^.*TARGETCORE_VERSION_MAJOR[ 	]+([0-9]+).*$" "\\1" _pm_ver_major "${_pm_ver_major}")
string(REGEX REPLACE "^.*TARGETCORE_VERSION_MINOR[ 	]+([0-9]+).*$" "\\1" _pm_ver_minor "${_pm_ver_minor}")
string(REGEX REPLACE "^.*TARGETCORE_VERSION_PATCH[ 	]+([0-9]+).*$" "\\1" _pm_ver_patch "${_pm_ver_patch}")
set(TARGETCORE_VERSION "${_pm_ver_major}.${_pm_ver_minor}.${_pm_ver_patch}")

install(TARGETS targetcore EXPORT TargetCoreTargets
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}      # .dll  (Windows)
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}      # .so   (Linux)
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})     # .lib import library
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/TargetCore_c.h
              ${CMAKE_CURRENT_SOURCE_DIR}/TargetCore_version.h
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
# The licence and the attribution notice travel with the binary: Apache-2.0
# section 4 requires the consumer receive both, and an install that drops them
# makes every downstream redistribution non-compliant. The NOTICE files of the
# three components are DIFFERENT files -- checked, not assumed -- so each goes
# to its own directory and none overwrites another.
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/LICENSE
              ${CMAKE_CURRENT_SOURCE_DIR}/NOTICE
        DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/targetcore)

install(EXPORT TargetCoreTargets
        FILE TargetCoreTargets.cmake
        NAMESPACE TargetCore::
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TargetCore)
configure_package_config_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/cmake/TargetCoreConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/TargetCoreConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TargetCore)
# SameMinorVersion, changed from SameMajorVersion at 0.10.0, and the change is
# the versioning policy made executable rather than a preference.
#
# The policy says a MINOR bump may break the covered C ABI while MAJOR is 0,
# and a PATCH may not. SameMajorVersion contradicted that outright: with MAJOR
# fixed at 0 it declared every 0.x compatible with every other, so
# find_package(TargetCore 0.9) would have been satisfied by a 0.11 that had
# REMOVED the symbol the consumer asked for -- and satisfied it silently, at
# configure time, leaving the failure to the linker. A rule the package file
# contradicts is not a rule.
#
# SameMinorVersion means MAJOR and MINOR must match and PATCH may differ, which
# is exactly what section 4 promises. It becomes wrong the day this project
# reaches 1.0 and MINOR starts being additive-only; that is one of the things
# section 7's gate list has to remember to change.
write_basic_package_version_file(
    ${CMAKE_CURRENT_BINARY_DIR}/TargetCoreConfigVersion.cmake
    VERSION ${TARGETCORE_VERSION}
    COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/TargetCoreConfig.cmake
              ${CMAKE_CURRENT_BINARY_DIR}/TargetCoreConfigVersion.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/TargetCore)

# ---------------------------------------------------------------------------
# p2p_abisurface -- the covered C ABI is what the manifest says it is.
#
# The versioning policy (Stage 5 step 15) makes exactly one promise about this library:
# the 83 flat C symbols in .github/ci/abi-flat.manifest. A promise that nothing
# measures is a wish, and Msgcore reached that conclusion first -- its Readme
# says a surface nobody measures widens one accidental dllexport at a time.
# This is that mechanism for this component, and it had none until 0.10.0.
#
# The check is in TWO halves on purpose and this is the half that needs a
# compiler. check_repo_invariants.py compares the manifest to the HEADER with
# no build at all; this compares it to what the BUILT library exports. Neither
# subsumes the other: the header can promise a symbol that was never compiled
# in (a link error for the consumer, invisible to a source-only check), and the
# library can export a name nobody declared (a promise nobody decided to make,
# invisible to a header-only check).
#
# Runs on both platforms -- dumpbin /EXPORTS on Windows, nm -D on Linux -- and
# is NOT excluded from sanitised trees, unlike p2p_installtree below: it reads
# an export table and never links anything, so the build's flags are irrelevant
# to it. It is registered before that block's early return() for the same
# reason.
#
# Not labelled `security`. It is a redistribution gate, like p2p_installtree.
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    if(WIN32)
        # dumpbin ships beside cl.exe. Resolved here rather than searched for at
        # test time, so a missing one is a configure-time fact rather than a
        # red test that looks like an ABI change.
        get_filename_component(_p2p_cl_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
        set(_p2p_dumpbin_arg --dumpbin "${_p2p_cl_dir}/dumpbin.exe")
    else()
        set(_p2p_dumpbin_arg "")
    endif()
    add_test(NAME p2p_abisurface
             COMMAND ${Python3_EXECUTABLE}
                     ${CMAKE_CURRENT_SOURCE_DIR}/.github/ci/check_abi_exports.py
                     --library  $<TARGET_FILE:targetcore>
                     --manifest ${CMAKE_CURRENT_SOURCE_DIR}/.github/ci/abi-flat.manifest
                     ${_p2p_dumpbin_arg})
    set_tests_properties(p2p_abisurface PROPERTIES TIMEOUT 120)
else()
    # Said out loud rather than skipped silently. A gate that quietly does not
    # exist is the failure mode this plan keeps recording.
    message(STATUS "p2p_abisurface: NOT registered -- no Python 3 interpreter "
                   "found. The covered ABI surface is then checked only against "
                   "the header (check_repo_invariants.py), not against the "
                   "built library.")
endif()

# ---------------------------------------------------------------------------
# p2p_installtree -- the gate for everything above.
#
# Install rules are the one part of a build that a normal test suite cannot
# reach: every in-tree test already has the source tree, the sibling targets and
# every include path in scope, so it passes whatever the install() lines stage.
# The entire install tree could be an empty directory and every other test in
# the suite would still pass. (Stating that without a number on purpose: this
# comment carried "54/54" long after the suite stopped registering 54 tests,
# and a total quoted in a rationale goes stale exactly the way a total quoted
# in a document does.) That is why this exists and why it runs a SEPARATE cmake: it stages
# into a scratch prefix, configures a project that knows only
# find_package(TargetCore), builds it, runs it with the build tree scrubbed off
# PATH, and then deletes the staged sibling and requires the same binary to
# stop working. The reasoning for each phase is in the script.
#
# Registered from THIS file rather than MscsUnitTests/ because it tests this
# directory's install() rules and nothing else, and because it must keep working
# in a TargetCore + Msgcore checkout that has no test repository in it.
#
# $<CONFIG> is a generator expression so the same registration serves the
# multi-config Visual Studio generator and single-config Ninja: on Windows it
# becomes Debug or Release at ctest time and is passed to --install, --build and
# CMAKE_BUILD_TYPE alike; on Linux CMAKE_BUILD_TYPE is already fixed and the
# expression yields it.
#
# Not labelled `security`: nothing here is an attack surface. It is a
# redistribution gate, and it is the exit criterion for Stage 5 step 14.
#
# NOT REGISTERED AT ALL in a sanitised tree, and the guard is deliberate rather
# than defensive. The consumer is a separate CMake project built with the
# toolchain's defaults, so it carries no `-fsanitize`; a program without the
# ASan runtime cannot link libraries built with it, and the gate would go red
# for a reason that has nothing to do with what it measures. Skipping is the
# honest outcome because the install RULES are identical under a sanitiser --
# the same files to the same places -- so there is nothing here a sanitised run
# could tell us that the ordinary run has not. Passing the flags through to the
# consumer instead was the alternative and was rejected: it would make this the
# one test whose subject is the sanitiser configuration rather than the install
# tree. `ctest -L security` never selected it either way (it carries no label),
# so this only changes what a bare `ctest` does on a sanitiser build directory.
string(FIND "${CMAKE_CXX_FLAGS} ${CMAKE_C_FLAGS}" "sanitize" _p2p_san)
if(_p2p_san GREATER -1)
    message(STATUS "p2p_installtree: NOT registered -- this tree is built with "
                   "sanitizers and the install consumer is not. See the note in "
                   "TargetCore/CMakeLists.txt.")
# else() rather than return(): return() from here is return from the whole
# DIRECTORY, so it would skip every line added below it for the rest of this
# file's life, in sanitised trees only, with no error. Today the file happens
# to end at the next statement, which is the only reason that was harmless --
# and "harmless because nothing follows it yet" is a trap armed for whoever
# appends the next target. Scoping the skip to the one test it is about costs
# an indent and cannot mislead later.
else()
    add_test(NAME p2p_installtree
             COMMAND ${CMAKE_COMMAND}
                     -DBUILD_DIR=${CMAKE_BINARY_DIR}
                     -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/install-check
                     -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/install-check-$<CONFIG>
                     -DCONFIG=$<CONFIG>
                     -DGENERATOR=${CMAKE_GENERATOR}
                     -DPLATFORM=${CMAKE_GENERATOR_PLATFORM}
                     -DTOOLSET=${CMAKE_GENERATOR_TOOLSET}
                     -P ${CMAKE_CURRENT_SOURCE_DIR}/install-check/run_install_check.cmake)
    # RUN_SERIAL because the phases run a full cmake configure and build of
    # their own; sharing a machine with a -j ctest run makes the timing of a
    # failure meaningless and the build itself slower than the thing it tests.
    # 600 s because a cold Visual Studio configure of even a one-file project is
    # tens of seconds, and a gate that times out on a slow machine reports a
    # staging defect that is not there.
    set_tests_properties(p2p_installtree PROPERTIES TIMEOUT 600 RUN_SERIAL TRUE)
endif()
